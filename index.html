1 ติดตั้งเครื่องมือพื้นฐาน + ตั้งค่า SSH & Firewall AlmaLinux

รันบน Terminal ของ AlmaLinux (root)

# ติดตั้งแพ็กเกจที่จำเป็น
dnf install -y curl wget nano git tar dnf-plugins-core openssh-server

# เปิดบริการ SSH และอนุญาต Root Login
systemctl enable --now sshd
echo "PermitRootLogin yes" | sudo tee /etc/ssh/sshd_config.d/root.conf
systemctl restart sshd

# ปรับ SELinux และ Kernel IP Forwarding
setenforce 0
sed -i 's/^SELINUX=enforcing/SELINUX=permissive/' /etc/selinux/config
echo "net.ipv4.ip_forward = 1" | sudo tee -a /etc/sysctl.d/99-ipforward.conf
sysctl -p /etc/sysctl.d/99-ipforward.conf

# เปิดพอร์ตบน Firewall (80, 81, 443, 2022)
firewall-cmd --add-port=80/tcp --permanent
firewall-cmd --add-port=81/tcp --permanent
firewall-cmd --add-port=443/tcp --permanent
firewall-cmd --add-port=2022/tcp --permanent
firewall-cmd --reload

systemctl status sshd

2 ติดตั้ง Docker Engine & Docker Compose AlmaLinux

รันบน Terminal ของ AlmaLinux (root)

# เพิ่ม Docker Repo และติดตั้ง
dnf config-manager --add-repo=https://download.docker.com/linux/centos/docker-ce.repo
dnf install -y docker-ce docker-ce-cli containerd.io docker-compose-plugin
# หรือถ้าเจอปัญหาให้ใช้แบบนี้แทน:
dnf config-manager --add-repo https://download.docker.com/linux/centos/docker-ce.repo

# เปิดใช้งาน Docker
systemctl enable --now docker

docker --version

3 ติดตั้ง Nginx Proxy Manager (NPM) Docker

สร้าง Directory และ Compose File แล้วรัน Container

mkdir -p ~/nginx-proxy-manager
cd ~/nginx-proxy-manager
cat << 'EOF' > docker-compose.yml
version: '3.8'
services:
  app:
    image: 'jc21/nginx-proxy-manager:latest'
    restart: unless-stopped
    ports:
      - '80:80'
      - '81:81'
      - '443:443'
      - '2022:2022'
    volumes:
      - ./data:/data
      - ./letsencrypt:/etc/letsencrypt
EOF

# สั่งรัน NPM Container
docker compose up -d

docker ps

4 รัน Web Services ปลายทางสำหรับทดสอบ L7 Docker

สร้าง Container demo-web และ web2 แล้วเปิดเบราว์เซอร์ดู NPM UI

docker run -d -p 8080:80 --name demo-web nginx
docker run -d -p 8081:80 --name web2 nginx

จากนั้นเปิดเบราว์เซอร์ไปที่ → http://127.0.0.1:81

ค่าเริ่มต้น NPM UI: user = admin@example.com / pass = changeme (บังคับเปลี่ยนครั้งแรกเมื่อล็อกอิน)

เปิด Notepad แบบ Run as Administrator → File → Open... ไปที่ C:\Windows\System32\drivers\etc\hosts แล้วเพิ่มบรรทัดเหล่านี้

127.0.0.1 demo.test
127.0.0.1 web2.test
127.0.0.1 redirect.test

6 สร้าง SSL Certificate + ดึงไฟล์มา Windows SSL + SCP

สร้าง Self-Signed Certificate บน AlmaLinux ก่อน แล้วค่อยดึงออกมาไว้ที่ Downloads

# สร้าง Self-Signed Certificate เก็บไว้ที่ /tmp ก่อนดึงออก
openssl req -x509 -nodes -days 365 -newkey rsa:2048 \
  -keyout /tmp/demo.key -out /tmp/demo.crt \
  -subj "/CN=demo.test"

ดึงไฟล์ Key/Cert ลง Windows (รันใน PowerShell):

scp -P 2222 root@127.0.0.1:/tmp/demo.key $env:USERPROFILE\Downloads\demo.key
scp -P 2222 root@127.0.0.1:/tmp/demo.crt $env:USERPROFILE\Downloads\demo.crt

7 สร้างโฟลเดอร์ /web2 ใน Container Docker

สร้างโฟลเดอร์และไฟล์ index.html ภายใน Container web2

docker exec -it web2 bash
cd /usr/share/nginx/html
mkdir -p web2
echo '<h1>Hello from web2 folder inside container</h1>' > web2/index.html
exit

เช็กว่าสำเร็จหรือไม่:

docker exec -it web2 cat /usr/share/nginx/html/web2/index.html
